sun_event(d, lat, lon, tz_offset, zenith=90.833): n = (d - date(d.year, 1, 1)).days + 1 gamma = 2 * math.pi / 365 * (n - 1 + 12/24) eqtime = 229.18 * (0.000075 + 0.001868*math.cos(gamma) - 0.032077*math.sin(gamma) - 0.014615*math.cos(2*gamma) - 0.040849*math.sin(2*gamma)) decl = (0.006918 - 0.399912*math.cos(gamma) + 0.070257*math.sin(gamma) - 0.006758*math.cos(2*gamma) + 0.000907*math.sin(2*gamma) - 0.002697*math.cos(3*gamma) + 0.00148*math.sin(3*gamma)) lat_r = math.radians(lat) cos_ha = (math.cos(math.radians(zenith)) / (math.cos(lat_r)*math.cos(decl)) - math.tan(lat_r)*math.tan(decl)) if cos_ha > 1 or cos_ha < -1: return None, None, eqtime, math.degrees(decl) ha = math.degrees(math.acos(cos_ha)) solar_noon_utc = 720 - 4*lon - eqtime return (solar_noon_utc - 4*ha)/60 + tz_offset, (solar_noon_utc + 4*ha)/60 + tz_offset, eqtime, math.degrees(decl) def build_dataset(year, lat, lon, tz): rows = [] d = date(year, 1, 1); end = date(year, 12, 31) while d <= end: sr, ss, eot, decl = sun_event(d, lat, lon, tz, 90.833) cdr, cds, _, _ = sun_event(d, lat, lon, tz, 96) ndr, nds, _, _ = sun_event(d, lat, lon, tz, 102) adr, ads, _, _ = sun_event(d, lat, lon, tz, 108) rows.append({ "date": d, "sunrise": sr, "sunset": ss, "civil_dawn": cdr, "civil_dusk": cds, "naut_dawn": ndr, "naut_dusk": nds, "astro_dawn": adr, "astro_dusk": ads,
